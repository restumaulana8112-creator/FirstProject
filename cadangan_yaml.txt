name: C++ Auto Build and Run

on:
  push:
    branches: [ "main" ]

jobs:
  build:
    runs-on: windows-latest

    steps:
    - name: Checkout code
      uses: actions/checkout@v4

    - name: Setup MSVC
      uses: ilammy/msvc-dev-cmd@v1

    - name: Compile and Run
      shell: powershell
      run: |
        # Cari semua file .cpp
        $files = Get-ChildItem -Filter *.cpp
        
        foreach ($f in $files) {
          Write-Host "--- MENGOMPILASI: $($f.Name) ---"
          
          # Kompilasi menjadi file .exe dengan nama yang sama
          cl.exe /EHsc $($f.Name)
          
          $exeName = $f.BaseName + ".exe"
          
          if (Test-Path $exeName) {
            Write-Host "--- MENJALANKAN: $exeName ---"
            # Cara panggil exe yang paling aman di PowerShell GitHub
            & ".\$exeName"
            Write-Host "-----------------------------"
          } else {
            Write-Host "Gagal membuat $exeName"
            exit 1
          }
        }
